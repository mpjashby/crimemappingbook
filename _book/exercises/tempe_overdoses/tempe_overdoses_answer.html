# This script produces a map of concentrations of calls to opioid overdoses in
# Tempe, Arizona, since January 2020. Run in the crime_mapping workspace.

# PREPARE ----------------------------------------------------------------------

## Load packages ----
pacman::p_load(ggspatial, here, httr2, sf, sfhotspot, tidyverse)


## Load data ----

# All data layers should be harmonised to use EPSG:6404 since this uses metres
# and covers the central part of Arizona

# Download the original data into the workspace
request("https://mpjashby.github.io/crimemappingdata/tempe_opioid_calls.csv") |>
  req_perform(path = here("data", "raw", "tempe_opioid_calls.csv"))
request(
  "https://mpjashby.github.io/crimemappingdata/tempe_character_areas.gml"
) |>
  req_perform(path = here("data", "raw", "tempe_character_areas.gml"))

# Read local calls data, keep calls since January 2020 and transform coordinates
incidents <- here("data", "raw", "tempe_opioid_calls.csv") |>
  read_csv() |>
  janitor::clean_names() |>
  filter(incident_date >= ymd("2020-01-01")) |>
  st_as_sf(coords = c("longitude", "latitude"), crs = "EPSG:4326") |>
  st_transform("EPSG:6404")

# Read the local city boundary and use the same coordinate reference system
character_areas <- here("data", "raw", "tempe_character_areas.gml") |>
  read_sf() |>
  janitor::clean_names() |>
  st_transform("EPSG:6404") |>
  # Some of the character areas have very long names which would not fit easily
  # on the map, so we abbreviate them first
  mutate(
    label = case_when(
      area_number == 1 ~ "North Tempe",
      area_number == 2 ~ "Double Butte",
      area_number == 3 ~ "Downtown",
      area_number == 7 ~ "The Lakes",
      area_number == 8 ~ "South Tempe",
      # If the area number is not specified above, use the existing area name
      TRUE ~ area_name
    )
  )


# WRANGLE ----------------------------------------------------------------------

# Estimate call density on a 200-metre grid, then clip it to the city boundary.
# Some recorded calls fall outside the boundary; a warning may be produced.
# Clipping limits the displayed density surface to the city rather than
# interpreting those approximate coordinates as exact locations.
incidents_kde <- incidents |>
  hotspot_kde(
    grid = hotspot_grid(character_areas, cell_size = 150),
    bandwidth_adjust = 0.33,
    quiet = TRUE
  ) |>
  hotspot_isoband(breaks = 10, quiet = TRUE) |>
  hotspot_clip(character_areas, quiet = TRUE)


# VISUALISE --------------------------------------------------------------------

# Create map
tempe_opioid_map <- hotspot_map(
  incidents_kde,
  basemap_type = "cartolight",
  caption = str_glue(
    "Author: Joe Bloggs, Date produced: {today()}\n",
    "Calls and character areas: City of Tempe (CC BY)\n",
    "Locations are approximate"
  )
) +
  # Add character area boundaries
  geom_sf(data = character_areas, colour = "grey50", fill = NA) +
  # Add character area labels
  geom_sf_label(
    aes(label = str_wrap(label, 10)),
    data = character_areas,
    alpha = 0.5,
    fill = "white",
    lineheight = 1,
    linewidth = 0,
    size = 2.5
  ) +
  # Add scale bar
  annotation_scale(width_hint = 1 / 5, style = "ticks", location = "bl") +
  labs(
    title = "Calls for suspected opioid overdoses in Tempe",
    subtitle = str_glue(
      "Calls to emergency medical services, ",
      "{format(min(pull(incidents, 'incident_date')), '%b %Y')} to ",
      "{format(max(pull(incidents, 'incident_date')), '%b %Y')}"
    ),
    fill = "density of\ncalls"
  ) +
  theme(
    legend.title = element_text(size = 8),
    plot.caption = element_text(hjust = 0, size = 6),
    plot.title = element_text(size = 12),
    plot.subtitle = element_text(size = 9, margin = margin(6, 0, 6, 0))
  )

# Save map
ggsave(
  here("output", "tempe_opioid_calls.jpg"),
  plot = tempe_opioid_map,
  width = 2000,
  height = 2000,
  units = "px"
)

Step 1: Explain the script

1
Start by explaining that the script maps concentrations of calls for suspected opioid overdoses in Tempe since January 2020. Calls are not the same as confirmed overdoses or deaths, so it is important to describe the data accurately. See Organising code with comments.

pacman::p_load() loads the packages needed for this analysis, listed in alphabetical order with the tidyverse package last. ggspatial adds map elements such as scale bars; here allows us to specify where to find files in the workspace; httr2 downloads data from the internet; sf handles spatial objects; sfhotspot creates crime maps and estimates density; tidyverse provides tools for reading, wrangling and plotting data. Loading packages at the start makes it easy to check which tools the script needs. See Loading packages.

Step 6: Calculate density, create isobands and clip the layer

# WRANGLE ----------------------------------------------------------------------

19# Estimate call density on a grid, then clip it to the character areas.
# Some recorded calls fall outside the boundary; a warning may be produced.
# Clipping limits the displayed density surface to the city rather than
# interpreting those approximate coordinates as exact locations.
20incidents_kde <- incidents |>
21  hotspot_kde(
22    grid = hotspot_grid(character_areas, cell_size = 150),
23    bandwidth_adjust = 0.33,
24    quiet = TRUE
  ) |>
25  hotspot_isoband(breaks = 10, quiet = TRUE) |>
26  hotspot_clip(character_areas, quiet = TRUE)
19
The dataset metadata tells us the recorded locations are approximate, and some fall outside the character-area boundaries. The comments explain why a warning may appear. Clipping the surface limits the displayed area; it does not correct the original locations.
20
Start with incidents and store the finished density layer as incidents_kde.
21
hotspot_kde() estimates the density of calls in each cell in a grid. See Mapping crime density.
22
hotspot_grid() defines a grid covering the character areas. cell_size = 150 makes the cells 150 metres across because the CRS of the character_areas dataset uses metres. cell_size = 150 is smaller than we might use for other cities, but Tempe is geographically a relatively small city (Tempe is a large suburb of Phoenix). See Specifying a grid for kernel density estimation.
23
bandwidth_adjust = 0.33 reduces the default smoothing bandwidth to reveal more local detail. See Fine-tuning cell size and bandwidth.
24
quiet = TRUE suppresses the progress display that is otherwise produced by hotspot_kde(). Use FALSE if you want to see progress during a long calculation.
25
hotspot_isoband() converts the gridded density estimates into polygons called isobands. breaks = 10 divides the density range into ten equal-width bands, so each polygon represents a range of density values rather than an exact value at every location. Using isobands can sometimes make patterns clearer than directly mapping the result produced by hotspot_kde(). Convert to isobands before clipping: hotspot_isoband() needs the complete grid produced by hotspot_kde(). quiet = TRUE suppresses messages telling you what settings hotspot_isoband() has automatically selected. See Converting density layers to isobands.
26
hotspot_clip() removes the parts of the density surface outside the combined character areas, while retaining the surface across the internal area boundaries. That stops the map showing density estimates outside the area selected for display. See Clipping map layers.

Overlay the character-area boundaries in grey. fill = NA leaves the interiors of the character area polygons unfilled so that they do not obscure the density layer. See Adding more layers; Visual hierarchy.
33
geom_sf_label() adds the character-area names so that readers can identify places where calls are concentrated. It uses the same spatial layer as the boundaries. See Adding more layers.
34
aes() specifies that the labels on the map should come from the label column in the character_areas object. str_wrap(label, 10) wraps names onto lines with a target width of 10 characters, helping them fit within the areas. It wraps between words rather than cutting words in half.
35
The labels have partly transparent white backgrounds (alpha = 0.5 and fill = "white"), compact line spacing (lineheight = 1), no border (linewidth = 0) and small text (size = 2.5). These choices help readers read the labels while keeping the density layer visually prominent.
36
Add a scale bar using ggspatial. width_hint = 1 / 5 suggests its width, style = "ticks" chooses its appearance and location = "bl" places it at the bottom left. See Scale bars.
37

38
Describe the data as calls for suspected opioid overdoses, rather than confirmed overdose deaths.
39
pull() extracts the date column, and min() and max() find the earliest and latest retained call dates. format() displays each date using an abbreviated month (%b) and a four-digit year (%Y), giving a concise range such as Jan 2020 to Sep 2024. str_glue() inserts this range into the subtitle, so it describes the downloaded records rather than assuming they extend to the present.
40
Use a short legend title that explains the colours without requiring readers to know the term KDE. \n breaks it over two lines. See Changing the legend title.
